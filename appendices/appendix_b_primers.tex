\section*{Cycling Profile Schematics}

\begin{itemize}
  \item \textbf{Mitochondrial \gene{COI}:}
    \begin{enumerate}
      \item Initial denaturation: \SI{94}{\celsius} for \SI{180}{\second}.
      \item 35 cycles: Denaturation at \SI{94}{\celsius} for \SI{30}{\second}; Annealing at \SI{50}{\celsius} for \SI{45}{\second}; Elongation at \SI{72}{\celsius} for \SI{60}{\second}.
      \item Final extension: \SI{72}{\celsius} for \SI{420}{\second}.
      \item Soak: \SI{4}{\celsius} hold.
    \end{enumerate}

  \item \textbf{Mitochondrial \gene{16S rRNA}:}
    \begin{enumerate}
      \item Initial denaturation: \SI{94}{\celsius} for \SI{120}{\second}.
      \item 32 cycles: Denaturation at \SI{94}{\celsius} for \SI{30}{\second}; Annealing at \SI{52}{\celsius} for \SI{40}{\second}; Elongation at \SI{72}{\celsius} for \SI{50}{\second}.
      \item Final extension: \SI{72}{\celsius} for \SI{300}{\second}.
    \end{enumerate}

  \item \textbf{Nuclear Exon \gene{RAG1} (Touchdown PCR):}
    \begin{enumerate}
      \item Initial denaturation: \SI{95}{\celsius} for \SI{300}{\second}.
      \item 10 touchdown cycles: \SI{94}{\celsius} for \SI{45}{\second}, \SI{60}{\celsius} to \SI{51}{\celsius} (\SI{-1}{\celsius} per cycle) for \SI{45}{\second}, \SI{72}{\celsius} for \SI{90}{\second}.
      \item 28 standard cycles: \SI{94}{\celsius} for \SI{45}{\second}, \SI{55}{\celsius} for \SI{45}{\second}, \SI{72}{\celsius} for \SI{90}{\second}.
      \item Final extension: \SI{72}{\celsius} for \SI{600}{\second}.
    \end{enumerate}
\end{itemize}
